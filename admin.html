Skrip SQL Inisialisasi Supabase

Jalankan di Supabase Dashboard → SQL Editor → New Query

-- 1. Buat Tabel portfolio_media
create table if not exists public.portfolio_media (
  id uuid primary key default gen_random_uuid(),
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  type text not null check (type in ('photo', 'video')),
  title text not null,
  category text not null default 'landscape',
  media_url text not null,
  thumbnail_url text,
  drone text default 'DJI Mavic 3 Pro',
  altitude text default '120m AGL',
  specs text default '4K UHD 60fps',
  location text default 'Indonesia',
  description text,
  is_featured boolean default false,
  display_order integer default 0
);

-- 2. Aktifkan RLS & Kebijakan Akses Publik
alter table public.portfolio_media enable row level security;

create policy "Allow public read access for all"
  on public.portfolio_media for select using (true);

create policy "Allow all insert"
  on public.portfolio_media for insert with check (true);

create policy "Allow all update"
  on public.portfolio_media for update using (true);

create policy "Allow all delete"
  on public.portfolio_media for delete using (true);

-- 3. Buat Bucket Storage 'drone-media'
insert into storage.buckets (id, name, public)
values ('drone-media', 'drone-media', true)
on conflict (id) do update set public = true;

create policy "Public Access to Drone Media"
  on storage.objects for select using (bucket_id = 'drone-media');

create policy "Public Upload to Drone Media"
  on storage.objects for insert with check (bucket_id = 'drone-media');
